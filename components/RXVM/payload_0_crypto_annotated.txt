# fn@263 — Custom hash: xxH1 mix + murmur3 fmix32 (exported as G8)
# ═══════════════════════════════════════════════════════════════════════
#
# NOT standard Murmur3 or xxHash32. Hybrid construction:
#
# Mixing loop: uses xxHash32 PRIME32_1 (0x9E3779B1 = 2654435761)
#   as a per-byte multiply-after-XOR. This is structurally FNV-like,
#   not Murmur3 (which processes 4-byte blocks with rotations).
#
# Finalization: Murmur3's fmix32 avalanche, verbatim:
#   h ^= len
#   h ^= h >>> 16; h = imul(0x85EBCA6B, h)
#   h ^= h >>> 13; h = imul(0xC2B2AE35, h)
#   h ^= h >>> 16
#   return h >>> 0
#
# Seed: 0xDEADBEEF
#
# Probably chosen for bytecode compactness — no block buffering,
# no rotations, just a tight loop + linear finalization. The fmix32
# does the heavy lifting for bit distribution.
#
# JS equivalent:
#   function customHash(input) {
#       let h = 0xDEADBEEF;
#       for (const byte of new TextEncoder().encode(input))
#           h = Math.imul(0x9E3779B1, h ^ byte);
#       h ^= input.length;
#       h ^= h >>> 16; h = Math.imul(0x85EBCA6B, h);
#       h ^= h >>> 13; h = Math.imul(0xC2B2AE35, h);
#       h ^= h >>> 16;
#       return h >>> 0;
#   }
#
# Used for hashing identifiers (UA + SID for cookie key derivation).
# NOT used for crypto key derivation — that goes through SHA-256.

# ═══════════════════════════════════════════════════════════════════════
#
# Textbook RC4. S-box initialized 0-255, permuted by key (KSA),
# then each input byte XOR'd with S[(S[i] + S[j]) % 256] (PRGA).
#
# This is the RC4 fallback path — used when WebCrypto's AES-CBC
# isn't available (old browsers, msCrypto).
#
# JS equivalent:
#   function rc4(data, key) {
#       const S = []; const out = [];
#       let i = 0, j = 0;
#       for (let k = 0; k < 256; k++) S[k] = k;
#       for (let k = 0; k < 256; k++) {
#           j = (j + S[k] + key[k % key.length]) % 256;
#           [S[j], S[k]] = [S[k], S[j]];
#       }
#       i = j = 0;
#       for (const byte of data) {
#           i = (i + 1) % 256;
#           j = (j + S[i]) % 256;
#           [S[i], S[j]] = [S[j], S[i]];
