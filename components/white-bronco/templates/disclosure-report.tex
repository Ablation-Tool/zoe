{

        description = "Detects vulnerable [component] in [Product] <= [version]"

    strings:
        // Version anchor: scopes match to affected builds
        $version = "[version string]" ascii

        // Byte pattern at or near the vulnerable function entry point
        $vuln_fn = { 55 48 89 E5 48 83 EC ?? 48 89 BD ?? ?? FF FF }

        // String from the vulnerable code path (optional second anchor)

    condition:
        $version and ($vuln_fn or $marker)
}
